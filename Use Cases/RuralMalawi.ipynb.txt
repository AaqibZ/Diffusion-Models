# =========================================================
# RuralMalawi DIFFUSION vs GROUND TRUTH (FRACTION-BASED, CLEAN)
# =========================================================

import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
from sklearn.metrics import mean_squared_error, mean_absolute_error

from usecasesruralmal import (
    models_k5,
    models_k10,
    models_k15,
    models_k20,
    ground_truth5,
    ground_truth10,
    ground_truth15,
    ground_truth20
)

# -------------------------
# PARAMETERS
# -------------------------
TIME_BINS = 10
SEED_COUNTS = [5, 10, 15, 20]

# -------------------------
# SNAPSHOTS
# -------------------------
snapshots = np.arange(1, TIME_BINS + 1)

# -------------------------
# GROUND TRUTH
# -------------------------
ground_truths = {
    5:  ground_truth5,   # (use closest available GT if k=5 missing)
    10: ground_truth10,
    15: ground_truth15,
    20: ground_truth20
}

# -------------------------
# MODEL ORDER (FIXED — 19)
# -------------------------
MODEL_ORDER = [
    "SCM","irSIR","EVM","TAM","POE","WPM","STM","AtI",
    "IC","LT","SI","SIR","LND","PAM","VMID","MAT","UAD",
    "SEIR","TIC"
]

# -------------------------
# CURVES
# -------------------------
CURVES = {
    5:  models_k5,
    10: models_k10,
    15: models_k15,
    20: models_k20
}

# -------------------------
# PLOTTING
# -------------------------
fig, axes = plt.subplots(1, 4, figsize=(22, 5), sharey=True)

markers = [
    "o","s","^","v","D","x","+","*","p","h",
    "<",">","P","X","d","|","_","8","."
]

linestyle = "--"
results = {}

for ax, k in zip(axes, SEED_COUNTS):

    gt = ground_truths[k]

    # ---- Ground Truth ----
    ax.plot(
        snapshots,
        gt,
        linestyle="--",
        linewidth=3,
        color="black",
        label="Ground Truth",
        zorder=10
    )

    table_rows = []

    # ---- Models ----
    for i, (name, curve) in enumerate(zip(MODEL_ORDER, CURVES[k])):

        spread = curve[:TIME_BINS]

        ax.plot(
            snapshots,
            spread,
            linestyle=linestyle,
            marker=markers[i % len(markers)],
            markevery=1,
            alpha=0.85,
            linewidth=1.2,
            label=name
        )

        # ---- Metrics ----
        rmse = np.sqrt(mean_squared_error(gt, spread))
        mae  = mean_absolute_error(gt, spread)
        peak_err = abs(spread.max() - gt.max())
        final_err = abs(spread[-1] - gt[-1])
        final_frac = spread[-1]

        table_rows.append([
            name,
            rmse,
            mae,
            peak_err,
            final_err,
            final_frac
        ])

    results[k] = pd.DataFrame(
        table_rows,
        columns=[
            "Model",
            "RMSE",
            "MAE",
            "PeakError",
            "FinalError",
            "FinalActivatedFraction"
        ]
    )

    ax.set_title(f"k = {k}", fontsize=14, fontweight="bold")
    ax.set_xlabel("Snapshots")
    ax.grid(alpha=0.3)
    ax.set_xlim(0.8, 10.2)
    ax.set_ylim(0.3, 1.02)

axes[0].set_ylabel("Fraction of Activated Nodes", fontsize=12)

# -------------------------
# GLOBAL LEGEND
# -------------------------
handles, labels = axes[0].get_legend_handles_labels()
fig.legend(
    handles,
    labels,
    loc="upper center",
    ncol=10,
    frameon=True,
    fontsize=10
)

fig.tight_layout(rect=[0, 0, 1, 0.88])
plt.savefig("ruralmal_diffusion_results.pdf", format="pdf", bbox_inches="tight")
plt.show()

# -------------------------
# PRINT TABLES
# -------------------------
for k, df in results.items():
    print(f"\n===== RuralMalawi METRICS (k = {k}) =====")
    print(df.round(4))
